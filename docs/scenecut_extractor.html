132        """
133        Calculate scene cuts with ffmpeg.
134
135        Args:
136            threshold (float): Threshold (between 0 and 1)
137            progress (bool): Show a progress bar on stderr
138        """
139        if not (0 <= threshold <= 1):
140            raise RuntimeError("Threshold must be between 0 and 1")
141
142        temp_dir = tempfile.mkdtemp()
143        temp_file_name = posixpath.join(
144            temp_dir, "scenecut-extractor-" + os.path.basename(self.input_file) + ".txt"
145        )
146
147        logger.debug("Writing to temp file: " + temp_file_name)
148
149        try:
150            cmd = [
151                self.ffmpeg_path,
152                "-nostdin",
153                "-loglevel",
154                "error",
155                "-y",
156                "-i",
157                self.input_file,
158                "-vf",
159                r"select=gte(scene\,0),metadata=print:file="
160                + escape_path_for_ffmpeg_filters(temp_file_name),
161                "-an",
162                "-f",
163                "null",
164                os.devnull,
165            ]
166
167            logger.debug(
168                "Running ffmpeg command: " + " ".join([shlex.quote(c) for c in cmd])
169            )
170
171            ff = FfmpegProgress(cmd)
172            if progress:
173                with tqdm(total=100, position=1) as pbar:
174                    for p in ff.run_command_with_progress():
175                        pbar.update(p - pbar.n)
176            else:
177                for _ in ff.run_command_with_progress():
178                    pass
179
180            lines: list[str] = []
181            if os.path.isfile(temp_file_name):
182                with open(temp_file_name, "r") as out_f:
183                    lines = out_f.readlines()
184
185            frames: list[ScenecutInfo] = []
186            last_frame_info: dict = {}
187            for line in lines:
188                line = line.strip()
189                if line.startswith("frame"):
190                    if ret := re.match(
191                        r"frame:(?P<frame>\d+)\s+pts:(?P<pts>[\d\.]+)\s+pts_time:(?P<pts_time>[\d\.]+)",
192                        line,
193                    ):
194                        ret_matches = ret.groupdict()
195                        last_frame_info["frame"] = int(ret_matches["frame"])
196                        last_frame_info["pts"] = float(ret_matches["pts"])
197                        last_frame_info["pts_time"] = float(ret_matches["pts_time"])
198                    else:
199                        raise RuntimeError("Wrongly formatted line: " + line)
200                    continue
201
202                if line.startswith("lavfi.scene_score") and (splits := line.split("=")):
203                    if len(splits):
204                        last_frame_info["score"] = float(splits[1])
205                    else:
206                        raise RuntimeError("Wrongly formatted line: " + line)
207                    frames.append(cast(ScenecutInfo, last_frame_info))
208                    last_frame_info = {}
209
210            self.scenecuts = [f for f in frames if f["score"] >= threshold]
211
212        except Exception as e:
213            raise e
214        finally:
215            if os.path.isfile(temp_file_name):
216                logger.debug("Removing temp file: " + temp_file_name)
217                os.remove(temp_file_name)
